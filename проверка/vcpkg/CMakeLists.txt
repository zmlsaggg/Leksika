cmake_minimum_required(VERSION 3.20)
project(leksika_consumer LANGUAGES C CXX)
find_package(leksika CONFIG REQUIRED)
add_executable(consumer_c consumer.c)
target_link_libraries(consumer_c PRIVATE leksika::leksika)
target_compile_features(consumer_c PRIVATE c_std_11)
add_executable(consumer_cpp consumer.cpp)
target_compile_features(consumer_cpp PRIVATE cxx_std_17)
target_link_libraries(consumer_cpp PRIVATE leksika::leksika)
if(MSVC)
  target_compile_options(consumer_c PRIVATE /utf-8)
  target_compile_options(consumer_cpp PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME consumer_c COMMAND consumer_c)
add_test(NAME consumer_cpp COMMAND consumer_cpp)
